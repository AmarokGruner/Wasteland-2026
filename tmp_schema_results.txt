  CREATE TABLE users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            role TEXT NOT NULL DEFAULT 'player',
            player_id INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, token TEXT, last_login TIMESTAMP, reset_token TEXT, reset_token_expires TIMESTAMP, tutorial_step INTEGER DEFAULT -1, email TEXT,
            FOREIGN KEY (player_id) REFERENCES players(id)

=== Table: provinzen ===
  id (INTEGER)
  player_id (INTEGER)
  world_x (INTEGER)
  world_y (INTEGER)
  name (TEXT)
  created_at (TIMESTAMP)
  terrain (TEXT)

  SQL:
  CREATE TABLE provinzen (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        player_id INTEGER NOT NULL,
        world_x INTEGER NOT NULL,
        world_y INTEGER NOT NULL,
        name TEXT DEFAULT '',
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, terrain TEXT DEFAULT 'Grasland',
        UNIQUE(world_x, world_y),
        FOREIGN KEY (player_id) REFERENCES players(id))

  SQL:
  CREATE TABLE provinz_gebaeude (

  SQL:

        x INTEGER,
        y INTEGER,
        terrain_id INTEGER,
        owner_id INTEGER,
        is_ruin_province INTEGER DEFAULT 0,
